{{- /* The footer stamp, from data/opm/build.json (scripts/gen-stamp.sh): the version label, then the refs the version documents. An anchored version (site/versions.conf pins fixed refs) and a line version (resolved from release lines on every build) name them: "documents cli X, library L, core Y, catalog Z, operator W, opm S", each with a link to the commit whose tree was built and how it was found (the anchor, a line, a pin, an override and its reason) as the link title. A ref that is a SHA shows as seven hex characters. In a line version a tree read from a branch (docs main or release/...) shows that branch and its commit: "opm main 4e582b8" when the ref is the branch itself, "core v2.0.0-beta.1 (docs main f5c4463)" when the ref is the release the stamp names and the docs come from a branch head; a tree read at its tag is that tag, so the text is exact. A source = main version, and a fixture build, shows the commit of each repo the build read, resolved on the host ("none" for a root that is not its own git checkout). Called from the Hextra hook custom/footer.html with the page. */ -}} {{- $label := site.Version.Name -}} {{- range site.Params.opm.versions -}}{{- if eq .name site.Version.Name -}}{{- $label = .label | default .name -}}{{- end -}}{{- end -}} {{- $sources := dict -}} {{- $ver := dict -}} {{- with hugo.Data.opm -}}{{- with .build -}} {{- $sources = .sources | default dict -}} {{- with .versions -}}{{- with index . site.Version.Name -}}{{- $ver = . -}}{{- end -}}{{- end -}} {{- end -}}{{- end -}}
{{ $label }} {{ if or (eq $ver.kind "anchored") (eq $ver.kind "line") -}} {{- $refs := $ver.refs | default dict -}} {{- $shown := slice -}} {{- range slice "cli" "library" "core" "catalog_opm" "opm-operator" "opm" -}}{{- if index $refs . -}}{{- $shown = $shown | append . -}}{{- end -}}{{- end -}} documents {{- range $i, $repo := $shown -}} {{- $r := index $refs $repo -}} {{- $text := $r.ref -}} {{- if findRE `^[0-9a-f]{40}$` $r.ref -}}{{- $text = substr $r.ref 0 7 -}}{{- end -}} {{- $docs := $r.docs | default "" -}} {{- $branch := and (eq $ver.kind "line") (or (eq $docs "main") (hasPrefix $docs "release/")) -}} {{- $name := index (dict "catalog_opm" "catalog" "opm-operator" "operator") $repo | default $repo -}} {{- $url := printf "%s/%s/commit/%s" site.Params.opm.github $repo $r.sha -}} {{- $title := replaceRE `^([a-z]+):` "$1: " $r.how -}} {{- if $i }},{{ end }} {{ $name }} {{- if and $branch (eq $r.ref $docs) }} {{ $docs }} {{ substr $r.sha 0 7 }} {{- else if $branch }} {{ $text }} (docs {{ $docs }} {{ substr $r.sha 0 7 }}) {{- else }} {{ $text }} {{- end }} {{- end -}} {{- else -}} {{ with $sources -}} built from {{ range $repo := site.Params.opm.repos -}} {{- $sha := index $sources $repo | default "none" -}} {{ $repo }} {{ if findRE `^[0-9a-f]{40}$` $sha -}} {{ substr $sha 0 7 }} {{- else -}} {{ $sha }} {{- end }} {{ end -}} {{- end }} {{- end }}
