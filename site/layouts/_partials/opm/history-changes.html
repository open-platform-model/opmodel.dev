{{- /* The "Changes in " section of a Catalogs member page, as Markdown the adapter (catalogs/_content.gotmpl) appends to the page body, so the table of contents, the anchor and the Markdown output carry it. Takes {history, segment, label}: history is the page's params.catalog.history before it is serialized. Returns "" unless the member changed in this segment (C13 changes[]). One line per change, in file order, worded from C13's ops only; for a pair compared in "paths" mode one sentence first says what was not compared. Field changes live only here, never inside the spec block. */ -}} {{- $out := "" -}} {{- $h := .history -}} {{- if and $h $h.member -}} {{- with index $h.member.changes .segment -}} {{- $lines := slice -}} {{- range . -}} {{- $p := partial "opm/history-code.html" .path -}} {{- $from := cond (eq .from nil) "" (partial "opm/history-code.html" .from) -}} {{- $to := cond (eq .to nil) "" (partial "opm/history-code.html" .to) -}} {{- $line := "" -}} {{- if eq .op "added" -}}{{- $line = printf "%s added" $p -}} {{- else if eq .op "removed" -}}{{- $line = printf "%s removed" $p -}} {{- else if eq .op "presence" -}} {{- /* C13's wording: by the target, and for a field becoming regular by where it came from. */ -}} {{- $ps := slice "regular" "optional" "required" -}} {{- if or (not (in $ps .from)) (not (in $ps .to)) (eq .from .to) -}} {{- errorf "catalogs: presence change of %s %s from %q to %q is not between two different presences" $h.fqn .path .from .to -}} {{- else if eq .to "required" -}}{{- $line = printf "%s made required" $p -}} {{- else if eq .to "optional" -}}{{- $line = printf "%s made optional" $p -}} {{- else -}}{{- $line = printf "%s no longer %s" $p .from -}}{{- end -}} {{- else if eq .op "type" -}}{{- $line = printf "%s type changed from %s to %s" $p $from $to -}} {{- else if eq .op "default" -}} {{- if not $from -}}{{- $line = printf "%s now defaults to %s" $p $to -}} {{- else if not $to -}}{{- $line = printf "%s no longer has a default (was %s)" $p $from -}} {{- else -}}{{- $line = printf "%s default changed from %s to %s" $p $from $to -}}{{- end -}} {{- else if eq .op "ref" -}} {{- if $to -}}{{- $line = printf "%s now refers to %s" $p $to -}} {{- else -}}{{- $line = printf "%s no longer refers to %s" $p $from -}}{{- end -}} {{- else if eq .op "spec" -}}{{- $line = "The spec changed in a way the field list does not show." -}} {{- else -}}{{- errorf "catalogs: change op %q of %s is not one of C13's" .op $h.fqn -}} {{- end -}} {{- $lines = $lines | append (printf "- %s" $line) -}} {{- end -}} {{- $out = printf "\n\n## Changes in %s\n\n" $.label -}} {{- if eq $h.mode "paths" -}} {{- $out = printf "%sCompared by field paths only: these two builds were made by different docs-kit minors, so type, default and reference changes are not shown.\n\n" $out -}} {{- end -}} {{- $out = printf "%s%s\n" $out (delimit $lines "\n") -}} {{- end -}} {{- end -}} {{- return $out -}}
