{{- /* One Markdown file of the enhancements repository, made ready for the Enhancements section (enhancements/_content.gotmpl): its HTML comments and its first heading (the layout prints the title) removed. A Hugo shortcode delimiter fails the build, since Hugo would run it. Takes {text, path}, path naming the file for the error. */ -}} {{- $s := .text -}} {{- if or (strings.Contains $s "{{<") (strings.Contains $s "{{%") -}} {{- errorf "enhancements: %s holds a Hugo shortcode delimiter ({{< or {{%%)" .path -}} {{- end -}} {{- $s = replaceRE `(?s)` "" $s -}} {{- $s = replaceRE `\A\s*# [^\n]*\n` "" $s -}} {{- return $s -}}
