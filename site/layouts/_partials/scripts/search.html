{{- /* OPM override of Hextra's scripts/search.html (v0.13.0, pinned in overrides.sha256). Hextra splits search into a data plane, which defines window.hextraSearch, and the command-palette UI (search-dialog.js). This loads only OPM's data plane, assets/js/opm-pagefind.js, which reads this version's Pagefind bundle (//pagefind/, built after Hugo by scripts/build-all.sh), so results render in Hextra's own palette and never leave the version being read. Nothing is loaded from a CDN. */ -}} {{- if (site.Params.search.enable | default true) -}} {{- $type := site.Params.search.type | default "pagefind" -}} {{- if ne $type "pagefind" -}} {{- errorf `search type %q is not supported; this site uses "pagefind"` $type -}} {{- end -}} {{- $prefix := printf "%s.%s" .Site.Version.Name .Language.Lang -}} {{- $data := resources.Get "js/opm-pagefind.js" | resources.ExecuteAsTemplate (printf "%s.pagefind.js" $prefix) . -}} {{- $dialog := resources.Get "js/search-dialog.js" | resources.ExecuteAsTemplate (printf "%s.search-dialog.js" $prefix) . -}} {{- if hugo.IsProduction -}} {{- $data = $data | minify | fingerprint -}} {{- $dialog = $dialog | minify | fingerprint -}} {{- end -}} {{- end -}}
